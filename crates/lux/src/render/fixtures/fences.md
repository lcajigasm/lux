# Fences

```rust
fn main() {
    let name = "lux";
    if name.is_empty() {
        return;
    }
    println!("{name}");
    let mut count = 1;
}
```

```javascript
function main() {
    const name = "lux";
    if (name.length === 0) {
        return;
    }
    console.log(name);
    let count = 1;
}
```

```python
def main():
    name = "lux"
    if name == "":
        return
    count = 0
    while count < 2:
        count += 1
    return count
```
